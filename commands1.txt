# Example command file 1 - Basic process creation and management
# This file demonstrates fork, exit, and wait operations

# Fork 3 children from init (PID 1)
fork 1
sleep 50

fork 1
sleep 50

fork 1
sleep 50

# Wait a bit for children to exist
sleep 200

# Print the process table
ps

# Exit the first children
exit 2 0
sleep 100
exit 3 0
sleep 100

# Wait for children
wait 1 -1
sleep 100

wait 1 -1
sleep 100

# Print final state
ps
